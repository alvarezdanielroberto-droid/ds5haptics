cmake_minimum_required(VERSION 3.16)
project(ds5haptics C CXX)
# Runtime de C++ estático: el .exe funciona en un Windows recién instalado sin nada extra
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(DS5_SPEAKER "Parlante del mando por Bluetooth (descarga y compila libopus)" ON)
set(DS5_VERSION "dev" CACHE STRING "Versión que muestra el programa (ej. v1.0.0)")

if(WIN32)
  add_executable(ds5haptics src/main_windows.cpp)
  target_include_directories(ds5haptics PRIVATE src)
  target_link_libraries(ds5haptics PRIVATE hid setupapi winmm ole32 ws2_32 user32 advapi32 psapi shell32 gdi32)
  target_compile_definitions(ds5haptics PRIVATE DS5_VERSION="${DS5_VERSION}")
  if(MSVC)
    target_compile_options(ds5haptics PRIVATE /utf-8 /W4)
  else()
    target_link_options(ds5haptics PRIVATE -municode)
  endif()

  if(DS5_SPEAKER)
    # libopus (licencia BSD) se descarga de xiph.org la primera vez que se compila
    include(FetchContent)
    set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
    set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(OPUS_BUILD_SHARED_LIBRARY OFF CACHE BOOL "" FORCE)
    set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
    set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    # Si el respaldo incluye el código de libopus en deps/, se usa sin internet
    set(OPUS_LOCAL "${CMAKE_SOURCE_DIR}/deps/opus-1.5.2.tar.gz")
    if(EXISTS "${OPUS_LOCAL}")
      FetchContent_Declare(opus URL "${OPUS_LOCAL}")
    else()
      FetchContent_Declare(opus URL https://downloads.xiph.org/releases/opus/opus-1.5.2.tar.gz)
    endif()
    FetchContent_MakeAvailable(opus)
    target_link_libraries(ds5haptics PRIVATE opus)
    target_compile_definitions(ds5haptics PRIVATE DS5_WITH_OPUS)
  endif()
endif()
